# BACKUS-NAUR FORM (BNF) OF THE MONAC'S PREPROCESSOR GRAMMAR

# IN UNIT COMBINATION PARSERS
<group> ::= <group-part> | <group> <group-part>
<group-part> ::= <if-section> | <control-line> | <text-line> | @ <non-directive>
<if-section> ::= <if-group> <elif-groups>* <else-group>* <endif-line>

<if-group> ::= @ if <constant-expression> <new-line> <group>* 
             | @ ifdef <identifier> <new-line> <group>* 
             | @ ifndef <identifier> <new-line> <group>*

<elif-groups> ::= <elif-group> | <elif-groups> <elif-group>
<elif-group> ::= @ elif <constant-expression> <new-line> <group>*
<else-group> ::= @ else <new-line> <group>*
<endif-line> ::= @ endif <new-line>
<control-line> ::= @ define <identifier> <replacement-list> <new-line> 
                 | @ define <identifier>( <identifier-list>* ) <replacement-list> <new-line>
                 | @ define <idenfifier>( <identifier-list>, ... ) <replacement-list> <new-line>
                 | @ undef  <identifier> <new-line>
                 | @ <new-line>
<text-line> ::= <preprocessing-token>* <new-line>
<identifier-list> ::= <identifier> | EBNF_GROUP(<identifier-list> , <identifier>)
<replacement-list> ::= <preprocessing-token>*

# UNIT PARSERS
<preprocessing-token> ::= <identifier> | <pp-number> | <character-literal> |  <string-literal> | <preprocessing-op-or-punc>
<identifier> ::= EBNF_GROUP(<identifier-nondigit> | EBNF_GROUP(<identifier> <identifier-nondigit>) | <identifier> <digit>)+
<identifier-nondigit> ::= [a-zA-Z] | _
<digit> ::= [0-9]
<pp-number> ::= <integer> | <floating-point> | <negative-integer> | <negative-floating-point>
<negative-floating-point> ::= -<floating-point>
<floating-point> ::= <integer>.<integer> | .<integer> | <integer>. 
<negative-integer> ::= -<integer>
<integer> ::= <digit>+
<string-literal> ::= " <char-sequence>* "
<character-literal> ::= '<char-sequence>'
<char-sequence> ::= <character> | EBNF_GROUP(<char-sequence> <character>)
<character> ::= [^'\\\r\n] | <escape-sequence>
<escape-sequence> ::= \' | \" | \\ | \n | \t | \r

<preprocessing-op-or-punc> ::= { 
                              | } 
                              | [ 
                              | ] 
                              | # 
                              | ## 
                              | ( 
                              | ) 
                              | ; 
                              | : 
                              | ... 
                              | ? 
                              | . 
                              | + 
                              | * 
                              | / 
                              | ^ 
                              | & 
                              | | 
                              | ~ 
                              | ! 
                              | = 
                              | < 
                              | > 
                              | += 
                              | -= 
                              | *= 
                              | /= 
                              | %= 
                              | ^= 
                              | &= 
                              | |=
                              | << 
                              | >>
                              | <<= 
                              | >>= 
                              | ==
                              | !=
                              | <=
                              | >=
                              | &&
                              | ||
                              | ++
                              | --
                              | ,
